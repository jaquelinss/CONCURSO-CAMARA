/**
 * Sanitiza uma string JSON mal formatada (com newlines literais) antes de fazer o parse.
 * Tenta um parse direto primeiro, e só sanitiza se o primeiro falhar.
 * @param {string} str A string JSON bruta da API.
 * @returns {object} O objeto JSON parseado.
 * @throws {Error} Se o parse falhar mesmo após a sanitização.
 */
const safeJsonParse = (str) => {
    if (typeof str !== 'string') {
        // Se a entrada não for string (ex: já é um objeto), retorna.
        // Isso pode acontecer se a API mudar o formato de resposta.
        if (typeof str === 'object' && str !== null) return str;
        throw new Error("Input to safeJsonParse must be a string or object.");
    }
    
    // Tenta extrair o JSON (regex que o código já usa)
    const jsonMatch = str.match(/\[.*\]|\{.*\}/s);
    const stringToParse = jsonMatch ? jsonMatch[0] : str;

    try {
        // Tenta parsear a string extraída diretamente
        return JSON.parse(stringToParse);
    } catch (e1) {
        // Se falhar (e1), pode ser por causa de \n ou \f (do LaTeX) literais dentro das strings.
        try {
            // FIX: Remove a sanitização de \n, \r, \t que estava quebrando o JSON.
            // Apenas sanitiza os backslashes (ex: \f de \frac para \\f).
            // Isso deve corrigir o "Bad escaped character" (e1) e
            // evitar o "Unexpected token '\'" (e2) que a sanitização anterior causava.
            const sanitizedString = stringToParse
                .replace(/\\/g, "\\\\"); // Apenas escapa os backslashes
                
            return JSON.parse(sanitizedString);
        } catch (e2) {
            // Se falhar de novo (e2), provavelmente é por causa de newlines literais *dentro*
            // das strings, que .replace(/\\/g, "\\\\") não corrige.
            // Lançamos o e2, pois é um erro mais profundo do que o e1.
            console.error("safeJsonParse failed on e2 (likely literal newlines in string):", e2, "Sanitized string was:", stringToParse.replace(/\\/g, "\\\\"));
            throw e2; 
        }
    }
};

/**
 * Sanitiza uma string para ser usada com segurança dentro de HTML.
 * @param {string} str A string para sanitizar.
 * @returns {string} A string sanitizada.
 */
const sanitize = (str) => {
    if (typeof str !== 'string') return '';
    const map = {
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#039;'
    };
    return str.replace(/[&<>"']/g, (m) => map[m]);
};

/**
 * Gera o conteúdo HTML para um quiz de questões, retornando um array de blocos HTML.
 * @param {Array} questions Array de objetos de questão.
 * @param {Object} settings As configurações do quiz.
 * @returns {Array<string>} Um array de strings HTML, cada uma representando uma questão.
*/
const generateQuizHtml = (questions, settings) => {
    const headerHtml = `
        <h1 style="text-align: center; border-bottom: 1px solid #ccc; padding-bottom: 10px; margin-bottom: 20px;">Quiz de ${sanitize(settings.subject)}</h1>
    `;
    const questionBlocks = questions.map((q, i) => `
        <div style="margin-bottom: 20px;">
            <p style="font-weight: bold;">Questão ${i + 1}: ${sanitize(q.pergunta)}</p>
            <ul style="list-style-type: none; padding-left: 20px;">
                ${q.opcoes.map((opt, optIndex) => `<li><strong>${String.fromCharCode(97 + optIndex)})</strong> ${sanitize(opt)}</li>`).join('')}
            </ul>
        </div>
    `);
    return [headerHtml, ...questionBlocks];
};

/**
 * Exporta um array de blocos HTML para um arquivo PDF, controlando as quebras de página.
 * @param {Array<string|HTMLElement>} blocks Um array de strings HTML ou elementos DOM.
 * @param {string} fileName O nome do arquivo PDF a ser salvo.
 * @param {Function} setError Função para definir mensagens de erro.
 */
const exportHtmlToPdf = async (blocks, fileName, setError) => {
    if (!window.jspdf || !window.html2canvas) {
        setError("A biblioteca de exportação de PDF ainda não está pronta. Tente novamente em alguns segundos.");
        return;
    }

    const { jsPDF } = window.jspdf;
    const pdf = new jsPDF({
        orientation: 'portrait',
        unit: 'mm',
        format: 'a4'
    });

    const pdfWidth = pdf.internal.pageSize.getWidth();
    const pdfHeight = pdf.internal.pageSize.getHeight();
    const margin = 15;
    const contentWidthPx = (pdfWidth - 2 * margin) * (96 / 25.4); 

    let cursorY = margin;

    for (let i = 0; i < blocks.length; i++) {
        const block = blocks[i];
        let elementToRender;
        let tempDiv = null;

        if (typeof block === 'string') {
            tempDiv = document.createElement('div');
            tempDiv.style.position = 'absolute';
            tempDiv.style.left = '-9999px';
            tempDiv.style.top = '0px';
            tempDiv.style.width = `${contentWidthPx}px`;
            tempDiv.style.padding = '0';
            tempDiv.style.fontFamily = 'Helvetica, Arial, sans-serif';
            tempDiv.style.fontSize = '12pt';
            tempDiv.style.color = '#000000';
            tempDiv.style.backgroundColor = '#ffffff';
            tempDiv.innerHTML = block;
            document.body.appendChild(tempDiv);
            elementToRender = tempDiv;
        } else {
            // É um elemento DOM
            elementToRender = block;
        }

        await new Promise(resolve => setTimeout(resolve, 50));

        try {
            const canvas = await window.html2canvas(elementToRender, {
                scale: 2,
                useCORS: true,
                logging: false,
                backgroundColor: '#ffffff', // Garante fundo branco
            });

            const imgData = canvas.toDataURL('image/png');
            const imgWidth = canvas.width;
            const imgHeight = canvas.height;
            const ratio = imgWidth / (pdfWidth - 2 * margin);
            const imgHeightInPdf = imgHeight / ratio;

            if (cursorY + imgHeightInPdf > pdfHeight - margin && i > 0) {
                pdf.addPage();
                cursorY = margin;
            }

            pdf.addImage(imgData, 'PNG', margin, cursorY, pdfWidth - 2 * margin, imgHeightInPdf);
            cursorY += imgHeightInPdf + 5;
        } catch (e) {
            console.error("Erro ao renderizar bloco para PDF:", e);
            setError("Ocorreu um erro ao renderizar um bloco de conteúdo para o PDF.");
            break;
        } finally {
            if (tempDiv && document.body.contains(tempDiv)) {
                document.body.removeChild(tempDiv);
            }
        }
    }
    
    pdf.save(fileName);
};

const PracticeQuiz = ({ questions, theme, onClose, quizTitle = "Quiz de Treino" }) => {
    const [currentIndex, setCurrentIndex] = useState(0);
    const [selectedAnswer, setSelectedAnswer] = useState(null);
    const [isCorrect, setIsCorrect] = useState(null);
    const [score, setScore] = useState(0);
    const [showResults, setShowResults] = useState(false);

    const currentQuestion = questions[currentIndex];

    const handleAnswer = (option) => {
        if (selectedAnswer !== null) return;
        setSelectedAnswer(option);
        const correct = option === currentQuestion.correta;
        setIsCorrect(correct);
        if (correct) {
            setScore(score + 1);
        }
    };

    const handleNext = () => {
        if (currentIndex < questions.length - 1) {
            setCurrentIndex(currentIndex + 1);
            setSelectedAnswer(null);
            setIsCorrect(null);
        } else {
            setShowResults(true);
        }
    };
    
    const handleRestart = () => {
        setCurrentIndex(0);
        setSelectedAnswer(null);
        setIsCorrect(null);
        setScore(0);
        setShowResults(false);
    }

    if (showResults) {
        return (
            <div className="w-full text-center p-6 bg-white rounded-lg shadow-md mt-6">
                <h3 className="text-2xl font-bold mb-4">{quizTitle} - Finalizado!</h3>
                <p className="text-xl mb-6">Sua pontuação: <span className={`font-bold ${theme.accent}`}>{score}</span> de {questions.length}</p>
                <div className="space-x-4">
                    <button onClick={handleRestart} className={`py-2 px-5 font-bold rounded-lg ${theme.button}`}>
                        Tentar Novamente
                    </button>
                    <button onClick={onClose} className={`py-2 px-5 font-bold rounded-lg bg-gray-500 hover:bg-gray-600 text-white`}>
                        Fechar Quiz
                    </button>
                </div>
            </div>
        );
    }

    return (
        <div className="w-full p-6 bg-white rounded-lg shadow-md mt-6">
            <h3 className="text-2xl font-bold mb-4 text-center">{quizTitle}</h3>
            <p className="text-sm font-semibold mb-2 opacity-70">Questão {currentIndex + 1} de {questions.length}</p>
            <p className="text-lg font-semibold my-4">{currentQuestion.pergunta}</p>
            <div className="space-y-3">
                {currentQuestion.opcoes.map((option, index) => {
                    const isSelected = selectedAnswer === option;
                    const isCorrectOption = currentQuestion.correta === option;
                    let buttonClass = `${theme.option}`;
                    if (isSelected) {
                        buttonClass = isCorrect ? 'bg-green-500 text-white' : 'bg-red-500 text-white';
                    } else if (selectedAnswer !== null && isCorrectOption) {
                        buttonClass = 'bg-green-500 text-white';
                    }
                    return (
                        <button
                            key={index}
                            onClick={() => handleAnswer(option)}
                            disabled={selectedAnswer !== null}
                            className={`w-full text-left p-3 rounded-lg transition-all duration-300 border-2 ${theme.border} ${buttonClass} disabled:cursor-not-allowed`}
                        >
                            {option}
                        </button>
                    );
                })}
            </div>
            {selectedAnswer && (
                 <div className="mt-4 p-4 rounded-lg bg-gray-100">
                    <p className="text-sm">{currentQuestion.explicacao}</p>
                    <button onClick={handleNext} className={`w-full mt-4 py-2 font-bold rounded-lg ${theme.button}`}>
                        {currentIndex < questions.length - 1 ? 'Próxima' : 'Ver Resultados'}
                    </button>
                </div>
            )}
        </div>
    );
};

const parseLessonContent = (content) => {
    if (typeof content !== 'string') return [];
    // A regex foi ajustada para capturar o termo antes do primeiro ":" e o resto como explicação.
    // O 's' no final permite que o "." capture novas linhas na explicação.
    const regex = /\[EXPLICACAO\](.*?):(.*?)\[\/EXPLICACAO\]/gs;
    const parts = [];
    let lastIndex = 0;
    let match;

    while ((match = regex.exec(content)) !== null) {
        // Adiciona o texto antes da tag [EXPLICACAO]
        if (match.index > lastIndex) {
            parts.push({ type: 'text', content: content.substring(lastIndex, match.index) });
        }
        
        const term = match[1]?.trim();
        const explanation = match[2]?.trim();

        if (term && explanation) {
            parts.push({ type: 'term', term: term, explanation: explanation });
        }
        
        lastIndex = regex.lastIndex;
    }

    // Adiciona qualquer texto restante após a última tag
    if (lastIndex < content.length) {
        parts.push({ type: 'text', content: content.substring(lastIndex) });
    }
    return parts;
};

            const parent = range.commonAncestorContainer.parentElement;
            const isHighlighted = parent.classList.contains('highlighted-text');
            setHighlighter({ visible: true, top: rect.top - contentRect.top - 40, left: rect.left - contentRect.left + rect.width / 2, isHighlighted });
        } else {
            setHighlighter({ visible: false, top: 0, left: 0, isHighlighted: false });
        }
    }, []);

    const applyHighlight = (color) => {
        if (selectionRef.current) {
            const range = selectionRef.current;
            const parent = range.commonAncestorContainer.parentElement;
            if (parent.classList.contains('highlighted-text')) {
                parent.style.backgroundColor = color;
            } else {
                const span = document.createElement('span');
                span.className = 'highlighted-text';
                span.style.backgroundColor = color;
                try {
                    range.surroundContents(span);
                } catch (e) {
                    span.appendChild(range.extractContents());
                    range.insertNode(span);
                }
            }
            window.getSelection().removeAllRanges();
            setHighlighter({ visible: false, top: 0, left: 0, isHighlighted: false });
        }
    };

    const removeHighlight = () => {
        if (selectionRef.current) {
            const range = selection.current;
            const parent = range.commonAncestorContainer.parentElement;
            if (parent.classList.contains('highlighted-text')) {
                const text = parent.innerHTML;
                parent.replaceWith(document.createRange().createContextualFragment(text));
            }
            window.getSelection().removeAllRanges();
            setHighlighter({ visible: false, top: 0, left: 0, isHighlighted: false });
        }
    };

    useEffect(() => {
        const contentElement = contentRef.current;
        const handleClickOutside = (event) => {
            if (tooltipRef.current && !tooltipRef.current.contains(event.target) && !event.target.classList.contains('term-highlight')) {
                hideTooltip();
            }
        };
        if (contentElement) {
            document.addEventListener('mouseup', handleSelection);
            document.addEventListener('touchend', handleSelection);
            document.addEventListener('mousedown', handleClickOutside);
            return () => {
                document.removeEventListener('mouseup', handleSelection);
                document.removeEventListener('touchend', handleSelection);
                document.removeEventListener('mousedown', handleClickOutside);
            };
        }
    }, [handleSelection]);

    const handleAskLessonDoubt = async () => {
        if (!doubt) return;
        setIsAsking(true);
        setDoubtResponse("");
        setSubQuestions([]); // Limpa subquestões anteriores
        const lessonContext = currentLesson.secoes.map(s => `${s.subtitulo}: ${s.conteudo}`).join('\n');
        const doubtPrompt = `Com base no seguinte material de estudo sobre "${currentLesson.titulo}":\n${lessonContext}\n\nResponda a seguinte dúvida do aluno: "${doubt}". Formate sua resposta usando HTML para melhor legibilidade. Use tags <p>, <strong>, e <ul>/<li>. Não inclua <html>, <head>, ou <body>.`;
        
        const apiKey = "";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: prompt }] }]
                })
            });
            if (!response.ok) throw new Error("A API de dúvidas falhou.");
            const result = await response.json();
            setDoubtResponse(result.candidates?.[0]?.content?.parts?.[0]?.text || "<p>Não foi possível obter uma resposta.</p>");
        } catch (error) {
            setDoubtResponse("<p>Ocorreu um erro ao processar sua dúvida.</p>");
        } finally {
            setIsAsking(false);
        }
    };

    const handleGenerateSubQuestions = async () => {
        setIsGeneratingSubQuestions(true);
        setSubQuestionError(null);
        setSubQuestions([]);

        const prompt = `Com base no contexto da seguinte dúvida de um aluno sobre a aula de "${currentLesson.titulo}": "${doubt}", e a resposta fornecida: "${doubtResponse.replace(/<[^>]*>?/gm, '')}", gere ${subQuestionCount} questões de múltipla escolha com dificuldade '${subQuestionDifficulty}'. O objetivo é testar o entendimento do aluno sobre o tópico da dúvida. A resposta DEVE ser um array de objetos JSON, cada um com as chaves "pergunta", "opcoes" (um array de 4 strings), "correta" (a string exata da resposta correta) e "explicacao".`;
        const apiKey = "";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: prompt }] }],
                    generationConfig: { responseMimeType: "application/json" }
                })
            });
            if (!response.ok) throw new Error("A API de geração de sub-questões falhou.");
            const result = await response.json();
            const text = result.candidates[0].content.parts[0].text;
            // FIX: Use safeJsonParse para evitar erros de caracteres de escape
            let parsedData = safeJsonParse(text);
            setSubQuestions(parsedData);
        } catch (error) {
            setSubQuestionError(error.message);
        } finally {
            setIsGeneratingSubQuestions(false);
        }
    };

    const handleGeneratePracticeQuiz = async () => {
        setIsGeneratingQuiz(true);
        setQuizError(null);
        setPracticeQuiz(null);

        const quizPrompt = `Gere 5 questões de múltipla escolha de dificuldade '${quizDifficulty}' sobre o tópico "${currentLesson.titulo}". Cada questão deve ter 4 opções de resposta. A resposta DEVE ser um array de objetos JSON, cada um com as chaves "pergunta", "opcoes" (um array de 4 strings), "correta" (a string exata da resposta correta) e "explicacao". Para conteúdo de química, use a abreviação correta "mols" para o plural de mol em português.`;

        const apiKey = "";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;
        
        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: quizPrompt }] }],
                    generationConfig: { responseMimeType: "application/json" }
                })
            });
            if (!response.ok) throw new Error("A API de geração de quiz falhou.");
            const result = await response.json();
            const text = result.candidates[0].content.parts[0].text;
            // FIX: Use safeJsonParse para evitar erros de caracteres de escape
            let parsedData = safeJsonParse(text);
            setPracticeQuiz(parsedData);
        } catch (error) {
            setQuizError(error.message);
        } finally {
            setIsGeneratingQuiz(false);
        }
    };

    return (
        <div className="w-full max-w-4xl mx-auto p-4 md:p-8">
            <NavigationHeader onBackToSubject={onBackToSubject} onBackToGeneral={onBackToGeneral} />
            <div className="flex justify-center mb-4 p-1 bg-gray-200 rounded-lg">
                {lessonLevels.map(level => (
                    <button 
                        key={level}
                        onClick={() => generateLessonForLevel(level)}
                        className={`flex-1 py-2 px-4 text-sm font-semibold rounded-md transition-colors ${currentLevel === level ? `${theme.button}` : 'text-gray-600 hover:bg-gray-300'}`}
                    >
                        {level}
                    </button>
                ))}
            </div>

            {isLoadingLevel && <LoadingSpinner theme={theme} text={`Carregando aula ${currentLevel.toLowerCase()}...`} />}

            {currentLesson && !isLoadingLevel && (
                <div className={`${theme.cardFront} p-6 rounded-xl shadow-lg relative`}>
                    <div ref={contentRef}>
                        {highlighter.visible && <HighlighterPalette top={highlighter.top} left={highlighter.left} onHighlight={applyHighlight} onRemove={removeHighlight} isHighlighted={highlighter.isHighlighted} />}
                        {tooltip.visible && <div ref={tooltipRef} className="explanation-tooltip" style={{ top: tooltip.top, left: tooltip.left }}>{tooltip.content}</div>}
                        
                        <h2 className={`text-4xl font-bold mb-4 ${theme.accent}`}>{currentLesson.titulo}</h2>
                        <p className="text-lg italic mb-6">{currentLesson.introducao}</p>
                        <div className="space-y-6">
                            {currentLesson?.secoes?.map((section, index) => (
                                <div key={index} className="p-4 bg-white rounded-lg shadow">
                                    <h3 className="text-2xl font-semibold mb-2">{section.subtitulo}</h3>
                                    <div className="text-base leading-relaxed whitespace-pre-wrap">
                                        {parseLessonContent(section.conteudo).map((part, i) => 
                                            part.type === 'term' ? (
                                                <span key={i} className="term-highlight" onClick={(e) => toggleTooltip(e, part.explanation)}>{part.term}</span>
                                            ) : (
                                                <span key={i} dangerouslySetInnerHTML={{ __html: part.content.replace(/\n/g, '<br />') }} />
                                            )
                                        )}
                                    </div>
                                </div>
                            ))}
                        </div>
                    </div>
                    
                    <div className="mt-8 border-t-2 pt-6 space-y-4">
                        <h3 className="text-2xl font-semibold">Ferramentas de Estudo</h3>
                        <div className="flex flex-col sm:flex-row gap-4 items-end">
                             <div className="flex-1">
                                <label className="block text-sm font-medium mb-1">Nível do Quiz</label>
                                <div className="flex p-1 bg-gray-200 rounded-lg">
                                    {['Introdutório', 'Intermediário', 'Aprofundado'].map(level => (
                                        <button 
                                            key={level}
                                            onClick={() => setQuizDifficulty(level)}
                                            className={`flex-1 py-1 px-2 text-xs font-semibold rounded-md transition-colors ${quizDifficulty === level ? `${theme.button}` : 'text-gray-600 hover:bg-gray-300'}`}
                                        >
                                            {level}
                                        </button>
                                    ))}
                                </div>
                            </div>
                            <button onClick={handleGeneratePracticeQuiz} disabled={isGeneratingQuiz} className={`flex-1 flex items-center justify-center gap-3 py-3 font-bold rounded-lg ${theme.button} disabled:opacity-50`}>
                                <ClipboardListIcon />
                                {practiceQuiz ? 'Gerar Novo Quiz' : 'Gerar Quiz de Treino'}
                            </button>
                            <button onClick={() => onSavePdf(currentLesson, contentRef.current, 'lesson')} disabled={!isPdfExportReady} className="flex-1 py-3 font-bold rounded-lg bg-green-600 hover:bg-green-700 text-white flex items-center justify-center gap-2 disabled:opacity-50">
                                <DownloadIcon />
                                {isPdfExportReady ? 'Salvar Aula em PDF' : 'Carregando...'}
                            </button>
                        </div>
                        {isGeneratingQuiz && <LoadingSpinner theme={theme} text="Gerando seu quiz de treino..." />}
                        {quizError && <p className="text-red-500">Erro: {quizError}</p>}
                        {practiceQuiz && <PracticeQuiz questions={practiceQuiz} theme={theme} onClose={() => setPracticeQuiz(null)} quizTitle={`Quiz ${quizDifficulty}`}/>}
                    </div>

                    <div className="mt-8 border-t-2 pt-6">
                        <h3 className="text-2xl font-semibold mb-4">Ainda com dúvidas? Pergunte à IA</h3>
                        <div className="flex gap-2">
                            <input type="text" value={doubt} onChange={(e) => setDoubt(e.target.value)} placeholder="Digite sua pergunta sobre a aula aqui..." className={`flex-grow p-2 rounded-lg ${theme.border} border-2`} />
                            <button onClick={handleAskLessonDoubt} disabled={isAsking || !doubt} className={`py-2 px-4 font-bold rounded-lg ${theme.button} disabled:opacity-50`}>{isAsking ? "..." : "Perguntar"}</button>
                        </div>
                        {isAsking && <div className="flex justify-center items-center p-4"><div className={`w-6 h-6 border-2 border-dashed rounded-full animate-spin ${theme.border}`}></div></div>}
                        {doubtResponse && (
                            <div className="mt-4 p-3 bg-blue-100 border-l-4 border-blue-400 rounded-r-lg">
                                <div className="prose prose-blue max-w-none" dangerouslySetInnerHTML={{ __html: doubtResponse }} />
                                
                                <div className="mt-4 pt-4 border-t border-blue-200">
                                    <h4 className="font-semibold text-blue-800">Quer aprofundar o conhecimento?</h4>
                                    <div className="flex flex-col sm:flex-row gap-4 mt-2 items-end">
                                        <CustomInput label="Nº de Questões" type="number" value={subQuestionCount} onChange={(e) => setSubQuestionCount(e.target.value)} theme={theme} />
                                        <CustomSelect label="Dificuldade" value={subQuestionDifficulty} onChange={(e) => setSubQuestionDifficulty(e.target.value)} options={difficulties} theme={theme} />
                                        <button onClick={handleGenerateSubQuestions} disabled={isGeneratingSubQuestions} className={`py-3 px-5 font-bold rounded-lg ${theme.button} disabled:opacity-50 whitespace-nowrap`}>
                                            Gerar Questões
                                        </button>
                                    </div>
                                    {isGeneratingSubQuestions && <LoadingSpinner theme={theme} text="Gerando sub-questões..." />}
                                    {subQuestionError && <p className="text-red-500 mt-2">Erro: {subQuestionError}</p>}
                                    {subQuestions.length > 0 && <PracticeQuiz key={doubt} questions={subQuestions} theme={theme} onClose={() => setSubQuestions([])} quizTitle="Quiz da Dúvida" />}
                                </div>
                            </div>
                        )}
                    </div>
                </div>
            )}
        </div>
    );
};

const GenerateMoreOptions = ({ currentSettings, onGenerate, onCancel, theme }) => {
    const [newTopicSettings, setNewTopicSettings] = useState({
        topic: currentSettings.topic,
        subTopic: 'Todos',
        specificTopic: ''
    });

    const isEnemArea = Object.keys(ENEM_AREAS).includes(currentSettings.subject);
    const isRedacao = currentSettings.subject === 'Redação';

    const handleTopicChange = (e) => {
        const newTopic = e.target.value;
        setNewTopicSettings({ ...newTopicSettings, topic: newTopic, subTopic: 'Todos', specificTopic: '' });
    };
    
    const handleSubTopicChange = (e) => {
        setNewTopicSettings({ ...newTopicSettings, subTopic: e.target.value, specificTopic: '' });
    };
    
    const handleSpecificTopicChange = (e) => {
        setNewTopicSettings({ ...newTopicSettings, specificTopic: e.target.value });
    };

    const handleSubmit = () => {
        onGenerate(newTopicSettings);
    };

    const getTopicsForCurrentSettings = () => {
        if (!isRedacao && !isEnemArea) {
            return topicsBySubject[currentSettings.subject] || {};
        }
        if (isRedacao) {
             if (currentSettings.model === 'Flashcard') return topicsBySubject.Redação.flashcardTopics;
            if (currentSettings.model === 'Enem') return topicsBySubject.Redação.proposalTopics;
        }
        if (isEnemArea) {
             return { 'Assuntos Mais Cobrados no ENEM': [] };
        }
        return {};
    };

    const currentTopics = getTopicsForCurrentSettings();
    const availableTopics = currentTopics ? ['Todos', ...Object.keys(currentTopics)].filter((v, i, a) => a.indexOf(v) === i) : [];
    const availableSubTopics = newTopicSettings.topic !== 'Todos' && currentTopics[newTopicSettings.topic] && currentTopics[newTopicSettings.topic].length > 0
        ? ['Todos', ...currentTopics[newTopicSettings.topic]]
        : [];
        
    const disableTopicSelection = !!newTopicSettings.specificTopic || isEnemArea;

    return (
        <div className="mt-8 p-6 bg-black/5 rounded-lg border border-gray-300">
            <h3 className="text-xl font-bold mb-4 text-center">Gerar Novas Questões</h3>
            <div className="space-y-4">
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <CustomSelect 
                        label={isRedacao && currentSettings.model === 'Enem' ? "Eixo Temático" : "Tópico"}
                        value={newTopicSettings.topic} 
                        onChange={handleTopicChange} 
                        options={availableTopics} 
                        theme={theme}
                        disabled={disableTopicSelection}
                    />
                    <CustomSelect 
                        label="Subtópico" 
                        value={newTopicSettings.subTopic} 
                        onChange={handleSubTopicChange} 
                        options={availableSubTopics} 
                        theme={theme}
                        disabled={disableTopicSelection || !availableSubTopics || availableSubTopics.length <= 1}
                    />
                </div>
                <CustomInput 
                    label="Ou digite um assunto específico"
                    value={newTopicSettings.specificTopic}
                    onChange={handleSpecificTopicChange}
                    theme={theme}
                    placeholder="Ex: Revolução Francesa"
                    disabled={isEnemArea}
                />
            </div>
            <div className="flex gap-4 mt-6 justify-end">
                <button onClick={onCancel} className="py-2 px-5 font-semibold rounded-lg bg-gray-400 hover:bg-gray-500 text-white transition-colors">Cancelar</button>
                <button onClick={handleSubmit} className={`py-2 px-5 font-bold rounded-lg ${theme.button} transition-colors`}>Gerar</button>
            </div>
        </div>
    );
};


const QuizScreen = ({ settings, questions, onBackToSubject, onBackToGeneral, onRegenerate, onCorrectEssay, correction, isCorrecting, theme, setError, isMammothReady, onSavePdf, isPdfExportReady, onGenerateMore }) => {
    const [currentIndex, setCurrentIndex] = useState(0);
    const [selectedAnswer, setSelectedAnswer] = useState(null);
    const [isCorrect, setIsCorrect] = useState(null);
    const [score, setScore] = useState(0);
    const [showResults, setShowResults] = useState(false);
    const [essayText, setEssayText] = useState("");
    const [essayImage, setEssayImage] = useState(null);
    const [textFileName, setTextFileName] = useState("");
    const [userEssayTheme, setUserEssayTheme] = useState("");
    const [tooltip, setTooltip] = useState(null);
    const correctedTextRef = React.useRef(null);
    const [doubt, setDoubt] = useState("");
    const [doubtResponse, setDoubtResponse] = useState("");
    const [isAsking, setIsAsking] = useState(false);
    const [essayDoubt, setEssayDoubt] = useState("");
    const [essayDoubtResponse, setEssayDoubtResponse] = useState("");
    const [isAskingEssayDoubt, setIsAskingEssayDoubt] = useState(false);
    const [eliminatedAnswers, setEliminatedAnswers] = useState(new Set());
    const [practiceQuestions, setPracticeQuestions] = useState([]);
    const [isGeneratingPractice, setIsGeneratingPractice] = useState(false);
    const [practiceQuestionCount, setPracticeQuestionCount] = useState(3);
    const [showMoreOptions, setShowMoreOptions] = useState(false);
    const [flashcardDoubt, setFlashcardDoubt] = useState("");
    const [flashcardDoubtResponse, setFlashcardDoubtResponse] = useState("");
    const [isAskingFlashcardDoubt, setIsAskingFlashcardDoubt] = useState(false);

    // Estados para as sub-questões
    const [subQuestions, setSubQuestions] = useState([]);
    const [isGeneratingSubQuestions, setIsGeneratingSubQuestions] = useState(false);
    const [subQuestionCount, setSubQuestionCount] = useState(3);
    const [subQuestionDifficulty, setSubQuestionDifficulty] = useState('Médio');
    const [subQuestionError, setSubQuestionError] = useState(null);

    const currentQuestion = questions[currentIndex];
    
    const handleImageUpload = (e) => {
        const file = e.target.files[0];
        if (file) {
            if (file.size > 4 * 1024 * 1024) { // 4MB limit
                setError("O arquivo de imagem é muito grande. O limite é de 4MB.");
                return;
            }
            setEssayText("");
            setTextFileName("");
            const reader = new FileReader();
            reader.onloadend = () => {
                setEssayImage(reader.result);
            };
            reader.readAsDataURL(file);
        }
    };
    
    const handleTextFileUpload = (e) => {
        const file = e.target.files[0];
        if (!file) return;

        if (file.size > 1 * 1024 * 1024) { // 1MB limit
            setError("O arquivo de texto é muito grande. O limite é de 1MB.");
            return;
        }

        setEssayImage(null);
        setTextFileName(file.name);
        const reader = new FileReader();

        if (file.name.endsWith('.docx')) {
            if (isMammothReady && window.mammoth) {
                reader.onload = (event) => {
                    window.mammoth.extractRawText({ arrayBuffer: event.target.result })
                        .then(result => {
                            setEssayText(result.value);
                        })
                        .catch(err => {
                            console.error("Error reading .docx file:", err);
                            setError("Não foi possível ler o arquivo .docx. Tente salvar como .txt e enviar novamente.");
                            setTextFileName("");
                        });
                };
                reader.readAsArrayBuffer(file);
            } else {
                 setError("O leitor de arquivos .docx ainda não está pronto. Tente novamente em alguns segundos ou use um arquivo .txt.");
                 setTextFileName("");
            }
        } else {
            reader.onload = (event) => {
                setEssayText(event.target.result);
            };
            reader.readAsText(file);
        }
    };

    const handleAnswer = (option) => {
        if (selectedAnswer !== null) return;
        setSelectedAnswer(option);
        const correct = option === currentQuestion.correta;
        setIsCorrect(correct);
        if (correct) {
            setScore(score + 1);
        }
    };

    const handleNext = () => {
        if (currentIndex < questions.length - 1) {
            setCurrentIndex(currentIndex + 1);
            setSelectedAnswer(null);
            setIsCorrect(null);
            setDoubtResponse("");
            setDoubt("");
            setEliminatedAnswers(new Set());
            setPracticeQuestions([]);
            setSubQuestions([]);
            if (settings.model === 'Flashcard') {
                setFlashcardDoubt("");
                setFlashcardDoubtResponse("");
            }
        } else {
            setShowResults(true);
        }
    };
    
    const handlePrevious = () => {
        if (currentIndex > 0) {
            setCurrentIndex(currentIndex - 1);
            setSelectedAnswer(null);
            setIsCorrect(null);
            setDoubtResponse("");
            setDoubt("");
            setEliminatedAnswers(new Set());
            setPracticeQuestions([]);
            setSubQuestions([]);
            if (settings.model === 'Flashcard') {
                setFlashcardDoubt("");
                setFlashcardDoubtResponse("");
            }
        }
    };

    const handleEliminateAnswer = (option) => {
        setEliminatedAnswers(prev => {
            const newSet = new Set(prev);
            if (newSet.has(option)) {
                newSet.delete(option);
            } else {
                newSet.add(option);
            }
            return newSet;
        });
    };
    
    const handleCorrectedTextClick = (e) => {
        const target = e.target;
        if (target.classList.contains('erro-vermelho') && target.title) {
            const rect = target.getBoundingClientRect();
            const containerRect = correctedTextRef.current.getBoundingClientRect();
            
            setTooltip({
                content: target.title,
                top: rect.top - containerRect.top - 10,
                left: rect.left - containerRect.left + rect.width / 2,
            });
        } else {
            setTooltip(null);
        }
    };
    
    const handleAskDoubt = async () => {
        if (!doubt || !currentQuestion) return;
        setIsAsking(true);
        setDoubtResponse("");
        setSubQuestions([]);
        
        const doubtPrompt = `Com base na seguinte questão do quiz: "${currentQuestion.pergunta}" e sua explicação: "${currentQuestion.explicacao}", responda a seguinte dúvida do aluno: "${doubt}". Formate sua resposta usando HTML para melhor legibilidade. Use tags <p> para parágrafos, <strong> para destacar termos importantes, e <ul>/<li> para listas, se necessário. Não inclua <html>, <head>, ou <body> tags.`;
        
        const apiKey = ""; // API Key is handled by the environment
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ 
                    contents: [{ parts: [{ text: doubtPrompt }] }] 
                })
            });

            if (!response.ok) {
                throw new Error("A API de dúvidas falhou em responder.");
            }

            const result = await response.json();
            if (result.candidates && result.candidates.length > 0) {
                const textResponse = result.candidates[0].content.parts[0].text;
                setDoubtResponse(textResponse);
            } else {
                setDoubtResponse("<p>Não foi possível obter uma resposta. Tente novamente.</p>");
            }
        } catch (error) {
            console.error("Error asking doubt:", error);
            setDoubtResponse("<p>Ocorreu um erro ao processar sua dúvida.</p>");
        } finally {
            setIsAsking(false);
        }
    };
    
    const handleGenerateSubQuestions = async () => {
        setIsGeneratingSubQuestions(true);
        setSubQuestionError(null);
        setSubQuestions([]);

        const prompt = `Com base no contexto da questão de quiz: "${currentQuestion.pergunta}", a explicação da resposta: "${currentQuestion.explicacao}", a dúvida do aluno: "${doubt}", e a resposta fornecida: "${doubtResponse.replace(/<[^>]*>?/gm, '')}", gere ${subQuestionCount} questões de múltipla escolha com dificuldade '${subQuestionDifficulty}'.                         const isEliminated = eliminatedAnswers.has(option);
                        let buttonClass = `${theme.option}`;

                        if (isSelected) {
                            buttonClass = isCorrect ? 'bg-green-500 text-white' : 'bg-red-500 text-white';
                        } else if (selectedAnswer !== null && isCorrectOption) {
                            buttonClass = 'bg-green-500 text-white';
                        }

                        return (
                            <div key={index} className="flex items-center gap-2">
                                <button
                                    onClick={() => handleAnswer(option)}
                                    disabled={selectedAnswer !== null}
                                    className={`w-full text-left p-4 rounded-lg transition-all duration-300 border-2 ${theme.border} ${buttonClass} disabled:cursor-not-allowed ${isEliminated ? 'line-through opacity-60' : ''}`}
                                >
                                    {option}
                                </button>
                                {selectedAnswer === null && (
                                    <button
                                        onClick={() => handleEliminateAnswer(option)}
                                        className={`p-2 rounded-full transition-colors ${isEliminated ? 'bg-gray-400 text-white' : 'bg-gray-200 hover:bg-gray-300'}`}
                                        aria-label={isEliminated ? "Restaurar alternativa" : "Eliminar alternativa"}
                                    >
                                        <BanIcon className="w-5 h-5" />
                                    </button>
                                )}
                            </div>
                        );
                    })}
                </div>

                {selectedAnswer && (
                    <div className={`mt-6 p-4 rounded-lg ${isCorrect ? 'bg-green-100 border-green-300' : 'bg-red-100 border-red-300'} border`}>
                        <div className="flex items-center gap-3 mb-2">
                            {isCorrect ? <CheckCircleIcon className="text-green-500" /> : <XCircleIcon className="text-red-500" />}
                            <h4 className={`text-lg font-bold ${isCorrect ? 'text-green-700' : 'text-red-700'}`}>{isCorrect ? 'Resposta Correta!' : 'Resposta Incorreta!'}</h4>
                        </div>
                        <p className="text-sm">{currentQuestion.explicacao}</p>
                        
                        <div className="mt-4">
                            <input 
                                type="text"
                                value={doubt}
                                onChange={(e) => setDoubt(e.target.value)}
                                placeholder="Ainda com dúvidas? Pergunte à IA"
                                className={`w-full p-2 rounded-lg ${theme.border} border-2`}
                            />
                            <button
                                onClick={handleAskDoubt}
                                disabled={isAsking || !doubt}
                                className={`w-full mt-2 py-2 font-bold rounded-lg ${theme.button} disabled:opacity-50`}
                            >
                                {isAsking ? "Pensando..." : "Perguntar"}
                            </button>
                            {doubtResponse && (
                                <div className="mt-4 p-3 bg-blue-100 border-l-4 border-blue-400 rounded-r-lg">
                                    <div className="prose prose-blue max-w-none" dangerouslySetInnerHTML={{ __html: doubtResponse }} />
                                    
                                    <div className="mt-4 pt-4 border-t border-blue-200">
                                        <h4 className="font-semibold text-blue-800">Quer aprofundar o conhecimento?</h4>
                                        <div className="flex flex-col sm:flex-row gap-4 mt-2 items-end">
                                            <CustomInput label="Nº de Questões" type="number" value={subQuestionCount} onChange={(e) => setSubQuestionCount(e.target.value)} theme={theme} />
                                            <CustomSelect label="Dificuldade" value={subQuestionDifficulty} onChange={(e) => setSubQuestionDifficulty(e.target.value)} options={difficulties} theme={theme} />
                                            <button onClick={handleGenerateSubQuestions} disabled={isGeneratingSubQuestions} className={`py-3 px-5 font-bold rounded-lg ${theme.button} disabled:opacity-50 whitespace-nowrap`}>
                                                Gerar Questões
                                            </button>
                                        </div>
                                        {isGeneratingSubQuestions && <LoadingSpinner theme={theme} text="Gerando sub-questões..." />}
                                        {subQuestionError && <p className="text-red-500 mt-2">Erro: {subQuestionError}</p>}
                                        {subQuestions.length > 0 && <PracticeQuiz key={doubt} questions={subQuestions} theme={theme} onClose={() => setSubQuestions([])} quizTitle="Quiz da Dúvida" />}
                                    </div>
                                </div>
                            )}
                        </div>
                         <div className="flex justify-between mt-6">
                            <button 
                                onClick={handlePrevious} 
                                disabled={currentIndex === 0}
                                className={`py-2 px-6 font-bold rounded-lg ${theme.button} disabled:bg-gray-400 disabled:cursor-not-allowed`}
                            >
                                Anterior
                            </button>
                            <button 
                                onClick={handleNext} 
                                className={`py-2 px-6 font-bold rounded-lg ${theme.button}`}
                            >
                                {currentIndex < questions.length - 1 ? 'Próxima Questão' : 'Ver Resultados'}
                            </button>
                        </div>
                    </div>
                )}
            </div>
        </div>
    );
};

A proposta deve ser relevante para a realidade brasileira. Forneça um título para a proposta, a frase-tema, 3 textos motivadores curtos (cada um com cerca de 50-80 palavras), e 4 sugestões de repertório sociocultural. A resposta DEVE ser um único objeto JSON com as chaves "tema", "frase_tema", "textos_motivadores" (um array de strings), e "repertorios" (um array de objetos, cada um com as chaves "tipo" [ex: "Filme", "Livro", "Citação", "Dado Histórico"] e "sugestao").`;

A resposta DEVE ser um array de objetos JSON, cada um com as chaves "pergunta", "opcoes" (um array de 4 strings), "correta" (a string exata da resposta correta) e "explicacao" (uma breve justificativa da resposta correta).`;

                    return false;
                }
            } else if (settings.model === 'Flashcard') {
                if (typeof item.frente !== 'string' || typeof item.verso !== 'string') return false;
            } else if (settings.model === 'Aula Explicativa') {
                if (typeof item.titulo !== 'string' || typeof item.introducao !== 'string' || !Array.isArray(item.secoes)) return false;
            } else {
                // Standard quiz question
                if (typeof item.pergunta !== 'string' || !Array.isArray(item.opcoes) || typeof item.correta !== 'string' || typeof item.explicacao !== 'string') {
                    return false;
                }
            }
        }
        return true;
    };

    const fetchWithRetry = async (url, options, retries = 4, delay = 1000) => {
        for (let i = 0; i < retries; i++) {
            try {
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), 60000); // 60s timeout per attempt

                const response = await fetch(url, { ...options, signal: controller.signal });
                clearTimeout(timeoutId);

                if (response.ok) {
                    return response;
                }
                
                if (response.status >= 400 && response.status < 500) {
                     const errorBody = await response.text();
                     throw new Error(`API error: ${response.statusText} - ${errorBody}`);
                }
                
                // Do not log retry attempts to the console to avoid clutter
            } catch (error) {
                if (i === retries - 1) throw error; 
            }
            await new Promise(res => setTimeout(res, delay));
            delay *= 2; 
        }
         throw new Error("API request failed after multiple retries.");
    };
    
    const callGeminiAPI = useCallback(async (prompt, imageBase64 = null) => {
        const isCorrection = prompt.includes("Você é um corretor de redações");

        if (!isCorrection) {
            setScreen('loading');
        } else {
            setIsCorrecting(true);
        }
        setError(null);
        if (isCorrection) setCorrection(null);
        
        const apiKey = ""; // API Key is handled by the environment
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

        try {
            const parts = [{ text: prompt }];
            if (imageBase64) {
                parts.push({
                    inlineData: {
                        mimeType: 'image/jpeg',
                        data: imageBase64.split(',')[1]
                    }
                });
            }

            const response = await fetchWithRetry(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts }],
                    generationConfig: {
                        responseMimeType: "application/json",
                        temperature: 0.2, 
                        topP: 0.95,
                        topK: 40
                    }
                }),
            });
            
            const result = await response.json();
            
            if (!result.candidates || result.candidates.length === 0) {
                throw new Error("A API não retornou uma resposta válida. Tente novamente.");
            }

            let text = result.candidates[0].content.parts[0].text;
            let parsedData;

            try {
                // FIX: Use safeJsonParse. 
                // Ele tentará o parse direto e, se falhar, sanitizará a string.
                parsedData = safeJsonParse(text);
            } catch (e) {
                console.error("JSON.parse failed:", e, "Raw text:", text);
                throw new Error("A IA retornou uma resposta em formato JSON inválido.");

            if (isCorrection) {
                if (!parsedData.nota_final || !parsedData.analise_competencias || !parsedData.texto_corrigido_html) {
                    throw new Error("A IA retornou uma correção em formato inesperado. Não foi possível exibir o feedback.");
                }
                setCorrection(parsedData);
            } else {
                let dataArray = Array.isArray(parsedData) ? parsedData : [parsedData];
                
                if (!validateQuizData(dataArray)) {
                    throw new Error("A IA retornou dados em um formato inesperado. Por favor, tente gerar o quiz novamente.");
                }

                setQuestions(dataArray);

                if (settings.model === 'Flashcard') {
                    const newFronts = dataArray.map(q => q.frente);
                    setPreviousFlashcards(prev => [...new Set([...prev, ...newFronts])]);
                }
                setScreen('quiz');
            }
        } catch (e) {
            console.error(e);
            setError(e.message || 'Ocorreu um erro. Tente novamente.');
            if (!isCorrection) {
                setScreen('settings');
            }
        } finally {
            if (isCorrection) {
                setIsCorrecting(false);
            }
        }
    }, [settings.model, settings.subject]);

            method: 'POST',
            headers: { 'Content-Type': 'application/json' },

                generationConfig: { responseMimeType: "application/json", temperature: 0.3 }
            })

            throw new Error("A API não retornou uma resposta válida para um dos tópicos.");

            const remainingQuestions = totalQuestions - foreignLangCount;
            if (remainingQuestions > 0) {
                const subjects = Object.keys(langConfig.distribution);
                let distributedCount = 0;
                subjects.forEach((subject, index) => {
                    let count;
                    if (index === subjects.length - 1) {
                        count = remainingQuestions - distributedCount;
                    } else {
                        count = Math.round(remainingQuestions * langConfig.distribution[subject]);
                        distributedCount += count;
                    }
                    if (count > 0) promptsConfig.push({ subject, quantity: count });
                });
            }
        } else {
            const subjects = Object.keys(areaConfig.distribution);
            let distributedCount = 0;
            subjects.forEach((subject, index) => {
                let count;
                if (index === subjects.length - 1) {
                    count = totalQuestions - distributedCount;
                } else {
                    count = Math.round(totalQuestions * areaConfig.distribution[subject]);
                    distributedCount += count;
                }
                if (count > 0) promptsConfig.push({ subject, quantity: count });
            });

                    successfulQuestions.push(...result.value);

            setScreen('quiz');

            console.error(e);
            setError(e.message || 'Ocorreu um erro ao gerar o quiz da área. Tente novamente.');
            setScreen('settings');
        }
    }, [settings, foreignLanguage, generatePrompt, fetchQuizChunk]);


    const generateQuiz = async (quizSettings) => {
        const isEnemArea = Object.keys(ENEM_AREAS).includes(quizSettings.subject);
        const isEnemTopic = quizSettings.topic === 'Assuntos Mais Cobrados no ENEM';
        const isEnemModel = quizSettings.model === 'Enem';

        if (isEnemArea && isEnemTopic && isEnemModel) {
            generateEnemAreaQuiz();

                    if (!validateQuizData(chunkQuestions)) {
                        throw new Error(`A IA retornou dados inválidos para uma parte do quiz.`);
                    }
                    allQuestions = [...allQuestions, ...chunkQuestions];
                }
                setQuestions(allQuestions);
                setScreen('quiz');
            } catch (e) {
                console.error(e);
                setError(e.message || 'Ocorreu um erro ao gerar o quiz em partes. Tente novamente.');
                setScreen('settings');
            }
        } else {
            if (quizSettings.subject === 'Redação' && quizSettings.model === 'Corrigir Redação Pronta') {
                setQuestions([{}]);
                setScreen('quiz');
                return;
            }
            setPreviousFlashcards([]);
            const prompt = generatePrompt(quizSettings);
            callGeminiAPI(prompt, null);
        }
    };

    const handleGenerateMore = (newTopicSettings = null) => {
        let newSettings = { ...settings };
        if (newTopicSettings) {
            newSettings = { ...newSettings, ...newTopicSettings };
        }
        setSettings(newSettings); // Atualiza o estado principal com as novas configurações
        setQuestions([]); // Limpa as questões antigas para sair da tela de resultados
        generateQuiz(newSettings);
    };
    
    const handleRegenerateFlashcards = () => {
        const prompt = generatePrompt(settings, true);
        callGeminiAPI(prompt, null);
    };

    const handleCorrectEssay = (essayText, essayImage, proposal) => {
        const textPrompt = essayText ? `A redação a ser corrigida é: "${essayText}"` : "A redação a ser corrigida está na imagem a seguir. Transcreva o texto e faça a análise.";
        const correctionPrompt = `Você é um corretor de redações especialista no modelo ENEM, baseado na matriz de correção oficial.
        Primeiro, se houver uma imagem, transcreva o texto dela com a máxima fidelidade. Preste atenção especial em reconhecer corretamente os acentos (como á, é, ç) e a pontuação. Seja extremamente cuidadoso com a pontuação, especialmente vírgulas. Respeite as vírgulas usadas corretamente pelo usuário e só aponte erro se a vírgula estiver comprovadamente mal empregada (ex: separando sujeito de predicado) ou ausente em um local obrigatório (ex: isolando um aposto). Se uma palavra estiver dividida no final de uma linha com um hífen, junte a palavra sem o hífen na transcrição.     const handleBackToSubjectSettings = () => {
        setScreen('settings');
        setQuestions([]);

    const handleBackToGeneralSettings = () => {
        setSettings(initialSettings);
        setScreen('settings');
        setQuestions([]);
        setError(null);
        setCorrection(null);
    };
    
    const handleSavePdf = async (data, liveElement, type) => {
        if (!isPdfExportReady) {
            setError("As bibliotecas de exportação de PDF ainda não estão prontas. Tente novamente em alguns segundos.");
            return;
        }

        if (type === 'flashcards') {
            try {
                const flashcardHtml = generateFlashcardsHtml(data, liveElement); // liveElement is settings here
                const tempDiv = document.createElement('div');
                tempDiv.innerHTML = flashcardHtml;
                document.body.appendChild(tempDiv);
                await new Promise(resolve => setTimeout(resolve, 300));
                
                const { jsPDF } = window.jspdf;
                const canvas = await window.html2canvas(tempDiv, { scale: 2, useCORS: true, logging: false });
                const imgData = canvas.toDataURL('image/png');
                const pdf = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });
                const pdfWidth = pdf.internal.pageSize.getWidth();
                const pdfHeight = pdf.internal.pageSize.getHeight();
                const imgWidth = canvas.width;
                const imgHeight = canvas.height;
                const ratio = imgWidth / pdfWidth;
                const canvasHeightInPdf = imgHeight / ratio;
                let position = 0;

                while (position < canvasHeightInPdf) {
                    if (position > 0) pdf.addPage();
                    pdf.addImage(imgData, 'PNG', 0, -position, pdfWidth, canvasHeightInPdf);
                    position += pdfHeight;
                }
                
                pdf.save(`Flashcards_${liveElement.subject.replace(/ /g, '_')}.pdf`);
                document.body.removeChild(tempDiv);
            } catch (e) {
                console.error("Erro ao gerar PDF de flashcards:", e);
                setError("Ocorreu um erro ao gerar o PDF dos flashcards.");
            }
            return;
        }

        let htmlBlocks = [];
        let fileName = 'estudo.pdf';

        try {
            switch (type) {
                case 'questions':
                    htmlBlocks = generateQuizHtml(data, liveElement); // liveElement is settings
                    fileName = `Quiz_${liveElement.subject.replace(/ /g, '_')}.pdf`;
                    break;
                case 'answers':
                    htmlBlocks = generateAnswersHtml(data, liveElement); // liveElement is settings
                    fileName = `Gabarito_${liveElement.subject.replace(/ /g, '_')}.pdf`;
                    break;
                case 'lesson':
                    const contentClone = liveElement.cloneNode(true);
                    contentClone.querySelectorAll('.explanation-tooltip, .text-highlighter').forEach(el => el.remove());
                    contentClone.querySelectorAll('.term-highlight').forEach(span => {
                        const strong = document.createElement('strong');
                        strong.textContent = span.textContent;
                        span.parentNode.replaceChild(strong, span);
                    });
                    
                    const glossaryBlock = generateGlossaryHtml(data);
                    htmlBlocks = [contentClone];
                    if (glossaryBlock) {
                        htmlBlocks.push(glossaryBlock);
                    }
                    fileName = `${data.titulo.replace(/ /g, '_')}.pdf`;
                    break;
                case 'correction':
                    htmlBlocks = [generateCorrectionHtml(data, liveElement)]; // liveElement is proposal
                    fileName = `Correcao_${liveElement.tema.replace(/ /g, '_')}.pdf`;
                    break;
                default:
                    setError("Tipo de PDF desconhecido.");
                    return;
            }
            await exportHtmlToPdf(htmlBlocks, fileName, setError);
        } catch (e) {
            console.error("Erro ao preparar HTML para PDF:", e);
            setError("Ocorreu um erro ao preparar o conteúdo para o PDF.");
        }
    };

    const renderScreen = () => {
        if (screen === 'settings') {
            return <SettingsScreen 
                onStart={() => generateQuiz(settings)} 
                theme={theme} 
                settings={settings} 
                setSettings={setSettings} 
                foreignLanguage={foreignLanguage}
                setForeignLanguage={setForeignLanguage}
            />;
        }
        
        if (screen === 'loading') {
            return <LoadingSpinner theme={theme} />;
        }

        if (screen === 'quiz' && questions && questions.length > 0) {
            if (settings.model === 'Aula Explicativa') {
                return <LessonScreen 
                    initialLesson={questions[0]} 
                    theme={theme} 
                    onBackToSubject={handleBackToSubjectSettings} 
                    onBackToGeneral={handleBackToGeneralSettings}
                    isPdfExportReady={isPdfExportReady}
                    setError={setError}
                    onSavePdf={handleSavePdf}
                    settings={settings}
                />
            }
            return <QuizScreen 
                settings={settings} 
                questions={questions} 
                onBackToSubject={handleBackToSubjectSettings}
                onBackToGeneral={handleBackToGeneralSettings}
                onRegenerate={handleRegenerateFlashcards} 
                onCorrectEssay={handleCorrectEssay}
                correction={correction}
                isCorrecting={isCorrecting}
                theme={theme} 
                setError={setError}
                isMammothReady={isMammothReady}
                onSavePdf={handleSavePdf}
                isPdfExportReady={isPdfExportReady}
                onGenerateMore={handleGenerateMore}
            />;
        }

        return null;
    };

    return (
        <main className={`min-h-screen w-full font-sans transition-colors duration-500 flex items-center justify-center p-4 ${theme.bg} ${theme.text}`}>
            <div className="relative w-full">
                {error && (
                    <div className="fixed top-5 left-1/2 -translate-x-1/2 bg-red-100 border border-red-400 text-red-700 px-4 py-3 rounded-lg shadow-lg z-50 flex items-center gap-4">
                        <p>{error}</p>
                        <button onClick={() => setError(null)} className="font-bold text-xl">&times;</button>
                    </div>
                )}
                {renderScreen()}
                 {settings.subject === 'Química' && screen === 'quiz' && (
                    <>
                        <button 
                            onClick={() => setShowPeriodicTable(!showPeriodicTable)}
                            className={`fixed bottom-5 left-5 w-16 h-16 rounded-full shadow-lg flex items-center justify-center z-40 transition-transform hover:scale-110 ${theme.button}`}
                            aria-label="Mostrar/Esconder Tabela Periódica"
                        >
                            <FlaskConicalIcon size={32} />
                        </button>
                        {showPeriodicTable && (
                            <div className="fixed top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 z-30">
                                <PeriodicTable onSelectElement={setHoveredElement} />
                            </div>
                        )}
                        <ElementDetailModal element={hoveredElement} theme={theme} />
                    </>
                )}
            </div>
        </main>
    );
}
